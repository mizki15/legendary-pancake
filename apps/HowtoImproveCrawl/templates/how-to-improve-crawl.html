はじめに：作業の全体像

更新するのは「設定」シートの小さな表、開始用の写真、Apps Script の3ファイルです。

1. 表を貼るB22:E26
2. 書き込み先を選ぶC22
3. 写真を設定startCrawl
4. コードを入れる3ファイル
最初にバックアップ（推奨・任意）
対象の Google シートで「ファイル → コピーを作成」を選び、元に戻せる状態にしてください。結果を書き込む「URL」シートの列や表の形は変えません。
1

「設定」シートの B22:E26 を貼る

切り替え先の Google シートで、対象の「設定」タブを開きます。下の表でそのタブを選び、「表をコピー」→ シートの B22 をクリック → 貼り付けの順に操作します。ほかの設定タブにも1つずつ繰り返します。

青い点線の C22 と C26 は、直接文字を直してからコピーできます。列記号 B～E と行番号 22～26 は位置の目印で、コピーには入りません。

各「設定」シートにあるスタート写真を右クリックして選び、写真の右上に出る ︙ を押します。メニューから 「スクリプトを割り当て」 を選び、下の文字だけを入力して OK を押します。

Apps Script の3つのコードを入れ替える

対象の Google シートで 「拡張機能 → Apps Script」 を開きます。同じプロジェクト内に次の3ファイルを用意し、各ファイルの内容を丸ごと入れ替えて保存します。下のコードは最初は閉じています。「表示」を押して確認でき、ボタンで全文をコピーできます。

  1. ファイルを選び、編集欄をクリックして Ctrl + A で中身をすべて選びます。
  2. このページで該当ファイルの「コードをコピー」を押し、編集欄に貼り付けます。
  3. 3ファイルとも終えたら、上部の保存ボタンを押します。古い A/B 専用の関数を別ファイルに残さないでください。

/** 複数の URL を最大 8 件ずつ並列取得する。失敗した URL だけ個別に再試行する。 */
function fetchHtmlBatch(urls, retryFunction) {
  var results = urls.map(function () { return {html: null, error: null}; });
  var indexes = [];
  var requests = [];
  for (var i = 0; i < urls.length; i++) {
    if (!urls[i]) continue;
    indexes.push(i);
    requests.push({url: urls[i], muteHttpExceptions: true, timeoutSeconds: 45});
  }
  if (!requests.length) return results;
  try {
    var responses = UrlFetchApp.fetchAll(requests);
    if (responses.length !== requests.length) throw new Error("一括取得の応答数が一致しません");
    for (var j = 0; j < responses.length; j++) {
      var index = indexes[j];
      var code = responses[j].getResponseCode();
      if (code >= 200 && code < 400) {
        results[index].html = responses[j].getContentText();
      } else {
        try {
          results[index].html = retryFunction(urls[index], 2, 1000);
        } catch (e) {
          results[index].error = e.message || String(e);
        }
      }
    }
  } catch (batchError) {
    for (var k = 0; k < indexes.length; k++) {
      var failedIndex = indexes[k];
      if (results[failedIndex].html !== null || results[failedIndex].error !== null) continue;
      try {
        results[failedIndex].html = retryFunction(urls[failedIndex], 3, 1000);
      } catch (e) {
        results[failedIndex].error = e.message || String(e);
      }
    }
  }
  return results;
}

/** 画像をクリックした時に表示中の設定シートを使う共通の入口。 */
function startCrawl() {
  var spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  var settingSheet = spreadsheet.getActiveSheet();
  if (!settingSheet || settingSheet.getRange("B22").getValue() !== "対象URLシート") {
    throw new Error("設定シート上の開始画像から実行してください。");
  }
  var target = resolveCrawlTarget(spreadsheet, settingSheet);
  if (stopCrawlerIfRequested(spreadsheet, settingSheet)) return;
  scheduleNextDailyCrawl(spreadsheet, settingSheet);
  runConfiguredCrawl(spreadsheet, settingSheet, target);
}

/** 時間トリガーには画像クリック時のシート情報がないため、triggerUid で復元する。 */
function resumeCrawl(e) {
  if (!e || !e.triggerUid) throw new Error("resumeCrawl は時間トリガーから実行してください。");
  var uid = String(e.triggerUid);
  var properties = PropertiesService.getScriptProperties();
  var key = "crawlTrigger:" + uid;
  var raw = properties.getProperty(key);
  if (!raw) return; // 既に停止・解除されたトリガー。
  var info = JSON.parse(raw);
  properties.deleteProperty(key);
  var triggers = ScriptApp.getProjectTriggers();
  for (var i = 0; i < triggers.length; i++) {
    if (String(triggers[i].getUniqueId()) === uid) ScriptApp.deleteTrigger(triggers[i]);
  }
  var spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  if (!spreadsheet || spreadsheet.getId() !== info.spreadsheetId) {
    throw new Error("トリガーの対象スプレッドシートを取得できません。");
  }
  var settingSheet = getSheetById(spreadsheet, info.sheetId);
  if (!settingSheet) throw new Error("トリガーの対象設定シートが見つかりません。");
  var target = resolveCrawlTarget(spreadsheet, settingSheet);
  if (stopCrawlerIfRequested(spreadsheet, settingSheet)) return;
  if (info.kind === "daily") scheduleNextDailyCrawl(spreadsheet, settingSheet);
  runConfiguredCrawl(spreadsheet, settingSheet, target);
}

function getSheetById(spreadsheet, sheetId) {
  var sheets = spreadsheet.getSheets();
  for (var i = 0; i < sheets.length; i++) {
    if (sheets[i].getSheetId() === Number(sheetId)) return sheets[i];
  }
  return null;
}

function resolveCrawlTarget(spreadsheet, settingSheet) {
  try {
    if (settingSheet.getRange("B22").getValue() !== "対象URLシート") {
      throw new Error(settingSheet.getName() + " は設定シートではありません。");
    }
    var urlSheetName = String(settingSheet.getRange("C22").getValue() || "").trim();
    if (!urlSheetName) throw new Error(settingSheet.getName() + "!C22 に URL シート名がありません。");
    var urlSheet = spreadsheet.getSheetByName(urlSheetName);
    if (!urlSheet) throw new Error("URL シートが見つかりません: " + urlSheetName);
    if (urlSheet.getRange("B1").getValue() !== "URL") {
      throw new Error(urlSheetName + " は URL シートの形式ではありません。");
    }
    var header = urlSheet.getRange("Q1").getValue();
    if (header !== "クロール往路" && header !== "県コード") {
      throw new Error(urlSheetName + " の Q1 から楽天・じゃらんを判別できません。");
    }
    return {sheet: urlSheet, kind: header === "クロール往路" ? "rakuten" : "jaran"};
  } catch (e) {
    setCrawlerStatus(settingSheet, "エラー", e.message || String(e), "-");
    throw e;
  }
}

function runConfiguredCrawl(spreadsheet, settingSheet, target) {
  if (target.kind === "rakuten") {
    crawlRakutenSetting(spreadsheet, settingSheet, target.sheet);
  } else {
    crawlJaranSetting(spreadsheet, settingSheet, target.sheet);
  }
}

/** 新しい設定シートを増やした後も、画像に同じ開始関数を割り当てられる。 */
function configureCrawlButtons() {
  var spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  var buttons = [];
  var sheets = spreadsheet.getSheets();
  for (var i = 0; i < sheets.length; i++) {
    var sheet = sheets[i];
    if (sheet.getRange("B22").getValue() !== "対象URLシート") continue;
    var images = sheet.getImages();
    var drawings = sheet.getDrawings();
    if (images.length + drawings.length !== 1) {
      throw new Error(sheet.getName() + " の開始画像・図形は1個にしてください。");
    }
    buttons.push(images.length ? {image: images[0]} : {drawing: drawings[0]});
  }
  for (var j = 0; j < buttons.length; j++) {
    if (buttons[j].image) buttons[j].image.assignScript("startCrawl");
    else buttons[j].drawing.setOnAction("startCrawl");
  }
  Logger.log(buttons.length + "枚の設定シートの画像を startCrawl に割り当てました。");
}

/** 設定 C26 の時刻で、同じ設定シートの翌日実行を予約する。空欄なら予約しない。 */
function scheduleNextDailyCrawl(spreadsheet, settingSheet) {
  var value = String(settingSheet.getRange("C26").getDisplayValue() || "").trim();
  if (!value) return;
  var match = /^(\d{1,2}):(\d{2})$/.exec(value);
  if (!match || Number(match[1]) > 23 || Number(match[2]) > 59) {
    throw new Error(settingSheet.getName() + "!C26 は HH:mm 形式で入力してください。");
  }
  var nextDate = new Date();
  nextDate.setDate(nextDate.getDate() + 1);
  nextDate.setHours(Number(match[1]), Number(match[2]), 0, 0);
  createCrawlTrigger(spreadsheet, settingSheet, "daily", nextDate);
}

function scheduleContinuation(spreadsheet, settingSheet) {
  createCrawlTrigger(spreadsheet, settingSheet, "continue", null);
}

function createCrawlTrigger(spreadsheet, settingSheet, kind, when) {
  deleteScheduledTriggersForSetting(spreadsheet, settingSheet, kind);
  var builder = ScriptApp.newTrigger("resumeCrawl").timeBased();
  var trigger = kind === "continue" ? builder.after(30 * 1000).create() : builder.at(when).create();
  var info = {spreadsheetId: spreadsheet.getId(), sheetId: settingSheet.getSheetId(), kind: kind};
  PropertiesService.getScriptProperties().setProperty("crawlTrigger:" + trigger.getUniqueId(), JSON.stringify(info));
}

function deleteScheduledTriggersForSetting(spreadsheet, settingSheet, kind) {
  var properties = PropertiesService.getScriptProperties();
  var triggers = ScriptApp.getProjectTriggers();
  for (var i = 0; i < triggers.length; i++) {
    if (triggers[i].getHandlerFunction() !== "resumeCrawl") continue;
    var key = "crawlTrigger:" + triggers[i].getUniqueId();
    var raw = properties.getProperty(key);
    if (!raw) continue;
    var info = JSON.parse(raw);
    if (info.spreadsheetId === spreadsheet.getId() &&
        Number(info.sheetId) === settingSheet.getSheetId() &&
        (!kind || info.kind === kind)) {
      ScriptApp.deleteTrigger(triggers[i]);
      properties.deleteProperty(key);
    }
  }
}

function setCrawlerStatus(settingSheet, status, detail, remainingText) {
  var colors = {
    "稼働中": {background: "#b7e1cd", foreground: "#137333"},
    "待機中": {background: "#fce8b2", foreground: "#8a5300"},
    "緊急停止": {background: "#f4c7c3", foreground: "#b31412"},
    "エラー": {background: "#f4c7c3", foreground: "#b31412"}
  };
  var color = colors[status] || colors["待機中"];
  settingSheet.getRange("C23").setValue(status).setBackground(color.background)
    .setFontColor(color.foreground).setFontWeight("bold");
  settingSheet.getRange("E23").setValue(detail);
  if (remainingText !== undefined) settingSheet.getRange("C25").setValue(remainingText);
}

function stopCrawlerIfRequested(spreadsheet, settingSheet) {
  if (String(settingSheet.getRange("C24").getValue() || "").trim() !== "緊急停止") return false;
  deleteScheduledTriggersForSetting(spreadsheet, settingSheet);
  setCrawlerStatus(settingSheet, "緊急停止", "停止指示を受け付けました。C24 を空欄にして手動で再開してください。", "停止");
  return true;
}

function estimateRemaining(startMs, startRow, completedRow, lastRow) {
  var processed = completedRow - startRow + 1;
  var remaining = Math.max(0, lastRow - completedRow);
  if (!remaining) return "0分";
  if (processed <= 0) return "計算中";
  var minutes = Math.max(1, Math.ceil((Date.now() - startMs) / processed * remaining / 60000));
  if (minutes < 60) return minutes + "分";
  return Math.floor(minutes / 60) + "時間" + (minutes % 60) + "分";
}

function updateCrawlerProgress(settingSheet, startMs, startRow, completedRow, lastRow) {
  settingSheet.getRange("E23").setValue(completedRow + " / " + lastRow + " 行まで完了");
  settingSheet.getRange("C25").setValue(estimateRemaining(startMs, startRow, completedRow, lastRow));
}

function onEdit(e) {
  if (!e || !e.range || e.range.getA1Notation() !== "C24") return;
  var settingSheet = e.range.getSheet();
  if (settingSheet.getRange("B22").getValue() !== "対象URLシート") return;
  if (String(e.range.getValue() || "").trim() === "緊急停止") {
    setCrawlerStatus(settingSheet, "緊急停止", "停止指示を受け付けました。実行中の取得が終わり次第停止します。", "停止");
  } else if (settingSheet.getRange("C23").getValue() === "緊急停止") {
    setCrawlerStatus(settingSheet, "待機中", "停止指示を解除しました。開始画像をクリックして再開してください。", "-");
  }
}

function crawlRakutenSetting(spreadsheet, sheetSetting, sheetUrl) {
  try {
  if (stopCrawlerIfRequested(spreadsheet, sheetSetting)) return;
  setCrawlerStatus(sheetSetting, "稼働中", "対象行を確認しています。", "計算中");
  var date = new Date();
  var startTime = new Date();

  // タイムスタンプ書き出し
  sheetSetting.getRange("F16").setValue(Utilities.formatDate(date, 'Asia/Tokyo', 'yy/MM/dd'));
  sheetSetting.getRange("F17").setValue(Utilities.formatDate(date, 'Asia/Tokyo', 'HH:mm'));

  // F18 は最後に完了した行。空欄なら見出しの次（2 行目）から始める。
  var startRowSetting = Math.max(2, Number(sheetSetting.getRange("C4").getValue()) + 1);
  var endRowSetting = sheetSetting.getRange("C5").getValue() + 1;

  var completedRow = Number(sheetSetting.getRange("F18").getValue()) || 0;
  var progressRow = Math.max(startRowSetting, completedRow + 1);

  var lastUsedRow = sheetUrl.getLastRow();
  var urlAColValues = lastUsedRow ? sheetUrl.getRange(1, 1, lastUsedRow, 1).getValues() : [];
  var lastDataRow = 0;
  for (var i = urlAColValues.length - 1; i >= 0; i--) {
    if (urlAColValues[i][0] !== "" && urlAColValues[i][0] != null) {
      lastDataRow = i + 1;
      break;
    }
  }

  var lp_start = progressRow;
  var lp_end = Math.min(lastDataRow, endRowSetting);

  if (lp_end < lp_start) {
    Logger.log("処理対象のデータがありません。");
    sheetSetting.getRange("F18").clearContent();
    setCrawlerStatus(sheetSetting, "待機中", "処理対象がありません。", "0分");
    return;
  }

  sheetUrl.getRange(lp_start, 20, lp_end - lp_start + 1, 7).clearContent();
  sheetUrl.getRange(lp_start, 3, lp_end - lp_start + 1, 2).clearContent();

  sheetSetting.getRange("Z2").setValue("");

  // 8 件ずつ並列取得し、シート更新はバッチ単位にまとめる。
  var batchSize = 8;
  var totalRows = lp_end - lp_start + 1;
  var inputValues = sheetUrl.getRange(lp_start, 2, totalRows, 9).getValues(); // B:J
  var stampDate = sheetSetting.getRange("F16").getValue();
  var stampTime = sheetSetting.getRange("F17").getValue();
  var errorLog = "";

  for (var offset = 0; offset < totalRows; offset += batchSize) {
    if (stopCrawlerIfRequested(spreadsheet, sheetSetting)) return;
    var count = Math.min(batchSize, totalRows - offset);
    var batchStart = lp_start + offset;
    var urls = [];
    for (var i = 0; i < count; i++) urls.push(inputValues[offset + i][0]);
    var fetched = fetchHtmlBatch(urls, fetchUrlWithRetryRakuten);
    if (stopCrawlerIfRequested(spreadsheet, sheetSetting)) return;
    var outputRows = [];
    var stampRows = [];

    for (var j = 0; j < count; j++) {
      var rowNumber = batchStart + j;
      var output = ["", "", "", "", "", "", ""]; // T:Z
      try {
        if (!urls[j]) throw new Error("URLが空です");
        if (fetched[j].error) throw new Error(fetched[j].error);
        var html = fetched[j].html;
        var pax = inputValues[offset + j][8]; // J
        var prices = [];
        if (pax == 1) {
          prices = extractWithRegex(html, /<span class="ndPrice">合計<strong>([\d,]+)<\/strong>/g);
        } else if (pax == 2) {
          prices = extractWithRegex(html, /<span class="prcPerPerson">\((?:1人あたり)?([\d,]+)円/g);
        }
        for (var p = 0; p < Math.min(prices.length, 3); p++) output[4 + p] = prices[p];

        var flights = extractFlights(html);
        for (var f = 0; f < Math.min(flights.length, 2); f++) output[f] = flights[f];
        var times = extractTimes(html);
        for (var t = 0; t < Math.min(times.length, 2); t++) output[2 + t] = times[t];
        stampRows.push([stampDate, stampTime]);
      } catch (e) {
        errorLog += "行" + rowNumber + "：" + e.message + "\n";
        Logger.log("行" + rowNumber + "：" + e.message);
        stampRows.push(["", ""]);
      }
      outputRows.push(output);
    }

    sheetUrl.getRange(batchStart, 20, count, 7).setValues(outputRows);
    sheetUrl.getRange(batchStart, 3, count, 2).setValues(stampRows);
    if (errorLog) sheetSetting.getRange("Z2").setValue(errorLog);
    sheetSetting.getRange("F18").setValue(batchStart + count - 1);
    if (offset % (batchSize * 5) === 0 || batchStart + count - 1 === lp_end) {
      updateCrawlerProgress(sheetSetting, startTime.getTime(), lp_start, batchStart + count - 1, lp_end);
    }

    if (batchStart + count - 1 < lp_end && new Date().getTime() - startTime.getTime() >= 4.5 * 60 * 1000) {
      if (stopCrawlerIfRequested(spreadsheet, sheetSetting)) return;
      scheduleContinuation(spreadsheet, sheetSetting);
      setCrawlerStatus(sheetSetting, "待機中", "定期的なクールダウン中。30秒後に続きを再開します。");
      Logger.log("実行時間の上限に近づいたため中断し、30秒後に再実行トリガーをセットしました。");
      return;
    }
  }

  if (stopCrawlerIfRequested(spreadsheet, sheetSetting)) return;

  try {
    var pt_sheet_val = sheetSetting.getRange("F14").getValue();
    var copy_cell_val = sheetSetting.getRange("K13").getValue();
    var pt_cell_val = sheetSetting.getRange("K14").getValue();

    var sheetCopyFrom = sheetUrl;
    var sheetCopyTo = spreadsheet.getSheetByName(pt_sheet_val);
    // このブックには履歴シートがないため、設定先が存在するときだけコピーする。
    if (sheetCopyFrom && sheetCopyTo && copy_cell_val && pt_cell_val) {
      var copyValue = sheetCopyFrom.getRange(copy_cell_val).getValues();
      sheetCopyTo.getRange(pt_cell_val).setValues(copyValue);
    } else {
      Logger.log("履歴コピーをスキップしました（参照先シートまたは範囲がありません）。");
    }
  } catch (e) {
    var errorLogCell = sheetSetting.getRange("Z2");
    var prevLog = errorLogCell.getValue();
    errorLogCell.setValue(prevLog + "履歴コピーエラー：" + e.message + "\n");
    Logger.log("履歴コピーエラー：" + e.message);
  }

  // 次回の定点実行は同じ URL 群の先頭から開始する。
  sheetSetting.getRange("F18").clearContent();
  setCrawlerStatus(sheetSetting, "待機中", "今回のクロールが完了しました。", "0分");
  } catch (e) {
    setCrawlerStatus(sheetSetting, "エラー", e.message || String(e), "-");
    throw e;
  }
}

function fetchUrlWithRetryRakuten(url, maxTries, retryDelayMs) {
  var tries = 0;
  while (tries < maxTries) {
    try {
      var response = UrlFetchApp.fetch(url, {timeoutSeconds: 45});
      return response.getContentText();
    } catch (e) {
      tries++;
      if (tries == maxTries) throw e;
      Utilities.sleep(retryDelayMs);
    }
  }
}

function extractWithRegex(html, regex) {
  var matches = [];
  var m;
  while ((m = regex.exec(html)) !== null) {
    matches.push(m[1].replace(/[^\d]/g, ""));
  }
  return matches;
}

function extractFlights(html) {
  var flights = [];
  var rows = html.match(/<tr class="active">[\s\S]*?<\/tr>/g) || [];
  for (var i = 0; i < rows.length; i++) {
    var m = rows[i].match(/<dt[^>]*>(.*?)<\/dt>[\s\S]*?<dd[^>]*>(.*?)<\/dd>/) || [];
    var carrier = m[1] ? m[1].trim() : "";
    var flightno = m[2] ? m[2].trim() : "";
    flights.push(carrier + " " + flightno);
  }
  return flights;
}

function extractTimes(html) {
  var times = [];
  var rows = html.match(/<tr class="active">[\s\S]*?<\/tr>/g) || [];
  for (var i = 0; i < rows.length; i++) {
    var m = rows[i].match(/(\d{1,2}:\d{2})⇒/);
    var time = m ? m[1] : "";
    times.push(time);
  }

function fetchUrlWithRetryJaran(url, maxTries, retryDelayMs){
  var tries = 0;
  while(tries < maxTries){
    try{
      var response = UrlFetchApp.fetch(url, {timeoutSeconds: 45});
      return response.getContentText();
    }catch(e){
      tries++;
      if(tries == maxTries) throw e;
      Utilities.sleep(retryDelayMs);
    }
  }
}

5

新しい表示と設定を使う

稼働状況 （C23 と E23）

緑：稼働中 黄：待機中 赤：停止・エラー
E23 に今の状況が文章で出ます。

緊急停止 （C24）

C24 に 緊急停止 と入力します。すぐに表示が赤くなり、処理は次の確認点で止まります。再開時は文字を消してから写真を押します。

予測残り時間 （C25）

終わった件数と経過時間から計算した目安です。通信の速さによって変わるため、確定した終了時刻ではありません。

翌日の実行時刻 （C26）

01:00 のように日本時間で入力します。空欄なら翌日分の予約を作りません。設定ごとに別の時刻を指定できます。

予約の作られ方
写真から startCrawl を実行すると、その設定の翌日分が予約されます。予約が呼ぶ関数名はどれも resumeCrawl ですが、予約ごとのIDで対象の設定シートを区別します。
6

切り替え時の注意事項

  1. 古いトリガーを確認： Apps Script 左側の時計アイコン「トリガー」を開き、旧コードのトリガーを整理してください。旧式と新式の両方が残ると二重実行の原因になります。
  2. resumeCrawl を手作業で追加しない： Apps Script の「トリガーを追加」画面で関数名だけ設定しても、どの「設定」シート用かを示す情報が作られません。日次予約は写真からの初回実行で作ります。
  3. 実行者を確認： 時間予約は作成した人のアカウントで動きます。別の人が同じ設定で予約を作ると重複する可能性があります。メール送信を使う処理では差出人も変わります。
  4. 停止は次の確認点： C24 の「緊急停止」は通信をその場で強制終了するものではありません。取得中の最大8件が終わるなど、次の確認点で止まります。
  5. 写真はパソコンのブラウザーから： Google シートのスマートフォンアプリでは、写真を押しても割り当てた処理は実行されません。
  6. 結果表の形を維持： この手順で貼る場所は「設定」シートの B22:E26 です。「URL」シートの見出しや列の並びは変更しません。
初回実行前に
4つの設定タブの C22 と C26、写真の割り当て、3ファイルの保存、旧トリガーの整理を確認してください。写真を押すと実際の取得と書き込みが始まります。
